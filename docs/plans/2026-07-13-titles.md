> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Render on-screen text (hook / ending / captions) into the reel — a typed `TITLES.json` the agent authors from `SPEC`, drawtext-rendered inside `deliver`'s single encode, with an opt-in curated font library resolved by name.

New `scripts/lib/fonts.py` (font resolution, extracted from draft so titles need not import it) + `scripts/lib/titles.py` (`Title`/`Titles` config-specs + `build_filterchain`) + `scripts/titles.py` (`build_titles` driver + CLI). `deliver` gains an optional `titles=` path that becomes a `-vf` drawtext chain in the encode it already runs; the drivers (`build_reel`, `edl_to_reel`) thread it through. Optional `montage.py` Stage; `setup.sh --with-fonts`.

**Tech Stack:** Python 3.13, dataclass config-specs (validate in `__post_init__`), ffmpeg `drawtext` (explicit `fontfile=` + per-title `textfile=` — the §12 landmines), pytest (real-ffmpeg golden with a drawtext skip-guard).

```python
"""Font resolution shared by the draft-preview labels (§5.9) and the titles render
(§5.16): the system-font fallback list, the friendly-name → cask-filename map, and
resolve_font. drawtext needs an explicit fontfile= — there is no reliable fontconfig
default on macOS (design §12). Extracted from draft.py so titles need not import the
draft-preview module."""
from pathlib import Path

FONT_CANDIDATES: tuple[str, ...] = (
    "/System/Library/Fonts/Supplemental/Arial.ttf",     # macOS system
    "/System/Library/Fonts/Helvetica.ttc",              # macOS system
    str(Path.home() / "Library/Fonts/DejaVuSans.ttf"),  # macOS, `brew install --cask font-dejavu`
    "/Library/Fonts/Arial.ttf",                         # macOS (older installs)
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",  # Debian/Ubuntu
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",           # Fedora/Arch
)

# friendly name -> the file `brew install --cask font-<x>` drops in ~/Library/Fonts
FONT_NAMES: dict[str, str] = {
    "Montserrat": "Montserrat-Regular.ttf",
    "Inter": "Inter-Regular.ttf",
    "Bebas Neue": "BebasNeue-Regular.ttf",
    "BebasNeue": "BebasNeue-Regular.ttf",
    "Oswald": "Oswald-Regular.ttf",
}

_SYSTEM = (None, "", "system")


def _system_default() -> str:
    for cand in FONT_CANDIDATES:
        if Path(cand).exists():
            return cand
    raise ValueError(
        f"no usable font found among {FONT_CANDIDATES} — pass a .ttf/.ttc path "
        "(drawtext needs an explicit fontfile=; there is no fontconfig default on "
        "macOS), or run `bash setup.sh` to install one")


def resolve_font(font: str | None = None) -> str:
    """A drawtext fontfile= path. `font`:
      None / "" / "system"  -> first existing FONT_CANDIDATES (system default)
      an explicit path       -> that path (must exist)
      a known NAME           -> ~/Library/Fonts/<cask file>, if installed
    Nothing usable → ValueError with the remedy. A NAMED font never silently falls
    back to a different one — that would change the creative output invisibly."""
    if font in _SYSTEM:
        return _system_default()
    looks_like_path = "/" in font or font.lower().endswith((".ttf", ".ttc", ".otf"))
    if looks_like_path:
        if Path(font).exists():
            return font
        raise ValueError(f"font path {font!r} does not exist — pass a real .ttf/.ttc/.otf")
    if font in FONT_NAMES:
        p = Path.home() / "Library/Fonts" / FONT_NAMES[font]
        if p.exists():
            return str(p)
        raise ValueError(
            f"font {font!r} is not installed at {p} — run `bash setup.sh --with-fonts` "
            "to install the curated set, or pass a .ttf/.ttc path")
    raise ValueError(
        f"font {font!r} is neither an existing path nor a known name {tuple(FONT_NAMES)} "
        "— pass a .ttf/.ttc/.otf path, or install & name one via `bash setup.sh --with-fonts`")
```

```python
"""Titles: on-screen text (hook / ending / captions) rendered into the reel at
deliver (design docs/specs/2026-07-13-titles-design.md, §5.16). A hard config-spec —
Title/Titles validate-and-raise in __post_init__, JSON round-trip, markdown preview —
like edl.py's EDL / spec.py's Spec. build_filterchain (Task 2) is the drawtext chain
deliver injects into its one encode. drawtext only; explicit fontfile= + per-title
textfile dodge the filtergraph-escaping landmines (§12)."""
import json
from dataclasses import dataclass
from pathlib import Path

from scripts.lib.fonts import resolve_font

POSITIONS: tuple[str, ...] = ("top", "center", "bottom")
STYLES: tuple[str, ...] = ("bold-box", "outline", "plain")


@dataclass(frozen=True)
class Title:
    text: str
    start: float
    dur: float
    font: str = "system"      # "system" -> resolve_font default; else a name/path (fonts.py)
    pos: str = "bottom"       # one of POSITIONS (safe-margin preset)
    style: str = "bold-box"   # one of STYLES

    def __post_init__(self) -> None:
        if not (isinstance(self.text, str) and self.text.strip()):
            raise ValueError(f"Title.text must be a non-empty string, got {self.text!r}")
        for name in ("start", "dur"):
            v = getattr(self, name)
            if isinstance(v, bool) or not isinstance(v, (int, float)):
                raise ValueError(f"Title.{name} must be a number (seconds), got {v!r}")
        if self.start < 0:
            raise ValueError(f"Title.start must be >= 0 seconds, got {self.start!r}")
        if self.dur <= 0:
            raise ValueError(f"Title.dur must be > 0 seconds, got {self.dur!r}")
        if not (isinstance(self.font, str) and self.font.strip()):
            raise ValueError(f"Title.font must be a non-empty name/path, got {self.font!r}")
        if self.pos not in POSITIONS:
            raise ValueError(f"Title.pos {self.pos!r} not in {POSITIONS}")
        if self.style not in STYLES:
            raise ValueError(f"Title.style {self.style!r} not in {STYLES}")

    def to_dict(self) -> dict:
        return {"text": self.text, "start": self.start, "dur": self.dur,
                "font": self.font, "pos": self.pos, "style": self.style}

    @classmethod
    def from_dict(cls, d: dict) -> "Title":
        def opt(key: str, default: object) -> object:
            v = d.get(key)
            return default if v is None else v
        return cls(text=d.get("text"), start=d.get("start"), dur=d.get("dur"),
                   font=opt("font", "system"), pos=opt("pos", "bottom"),
                   style=opt("style", "bold-box"))


@dataclass(frozen=True)
class Titles:
    items: tuple[Title, ...] = ()

    def __post_init__(self) -> None:
        if not all(isinstance(t, Title) for t in self.items):
            raise ValueError("Titles.items must all be Title instances")
        object.__setattr__(self, "items", tuple(self.items))

    def to_dict(self) -> dict:
        return {"items": [t.to_dict() for t in self.items]}

    @classmethod
    def from_dict(cls, d: dict) -> "Titles":
        return cls(items=tuple(Title.from_dict(t) for t in d.get("items", ())))

    @classmethod
    def from_json(cls, path: str | Path) -> "Titles":
        try:
            d = json.loads(Path(path).read_text())
        except json.JSONDecodeError as e:
            raise ValueError(f"{path} is corrupt ({e}) — re-run titles to regenerate it") from e
        return cls.from_dict(d)

    def write(self, path: str | Path) -> None:
        Path(path).write_text(json.dumps(self.to_dict(), indent=2))

    def to_markdown(self) -> str:
        lines = ["# Titles", "",
                 "| # | text | start→end | pos | style | font |",
                 "|---|------|-----------|-----|-------|------|"]
        for i, t in enumerate(self.items):
            lines.append(f"| {i} | {t.text.replace(chr(10), ' ')} | "
                         f"{t.start:.1f}→{t.start + t.dur:.1f} | {t.pos} | {t.style} | {t.font} |")
        if not self.items:
            lines.append("| – | (none) | – | – | – | – |")
        return "\n".join(lines)
```

```python
_STYLE_FRAGMENTS: dict[str, str] = {
    "bold-box": "box=1:boxcolor=black@0.5:boxborderw=16",
    "outline": "borderw=4:bordercolor=black",
    "plain": "",
}
_Y_EXPR: dict[str, str] = {          # safe-margin presets (design §4); ffmpeg h/th exprs
    "top": "h*0.08",
    "center": "(h-th)/2",
    "bottom": "h-th-h*0.12",
}


def build_filterchain(titles: Titles, work_dir: str, resolve=resolve_font) -> str:
    """The drawtext chain deliver injects into its encode -vf: one drawtext per Title,
    shown only in [start, start+dur] via enable=. Positions use ffmpeg's own w/h/tw/th
    expressions (resolution-agnostic — no python dims). Each title's free text goes to
    a per-title textfile (textfile= dodges filtergraph escaping of ':'/apostrophes —
    draft's trick, generalized to N titles, hence work_dir). Empty Titles -> "" (deliver
    then encodes with no -vf). font/textfile are single-quoted; a resolved system/cask
    font and a skill-controlled work-dir path carry no apostrophe (draft's assumption)."""
    if not titles.items:
        return ""
    work = Path(work_dir)
    work.mkdir(parents=True, exist_ok=True)
    parts: list[str] = []
    for i, t in enumerate(titles.items):
        tf = work / f"title_{i}.txt"
        tf.write_text(t.text)
        font = resolve(t.font)
        style = _STYLE_FRAGMENTS[t.style]
        piece = (f"drawtext=fontfile='{font}':textfile='{tf}':fontsize=h/18:"
                 f"fontcolor=white:x=(w-tw)/2:y={_Y_EXPR[t.pos]}:"
                 f"{style + ':' if style else ''}"
                 f"enable='between(t,{t.start},{t.start + t.dur})'")
        parts.append(piece)
    return ",".join(parts)
```

```python
"""Titles writer (design docs/specs/2026-07-13-titles-design.md, §5.16). The thin
mechanical half: from a Spec + the reel length, draft a Titles (hook opens, ending
closes, text as a mid caption) and write TITLES.json + TITLES.md. Judgment — the
actual words / timing / style — is the titles SKILL's; this is the starting
projection. Mirror of reel_brief.py: contract in lib, driver here."""
import argparse
import json
from pathlib import Path

from scripts.lib.spec import Spec
from scripts.lib.titles import Title, Titles


def build_titles(spec: Spec, total_s: float) -> Titles:
    """A sensible starting Titles from a Spec: the hook opens [0, 2], spec.text (when
    set) a mid caption, spec.ending (when set) a closing title in the last 3s. All
    font='system' so the draft renders without --with-fonts. total_s (reel length) is
    the caller's — EDL sum or a probed master — build_titles never guesses it."""
    if isinstance(total_s, bool) or not isinstance(total_s, (int, float)) or total_s <= 0:
        raise ValueError(f"build_titles total_s must be a positive number (seconds), got {total_s!r}")
    items: list[Title] = [Title(text=spec.hook, start=0.0, dur=min(2.0, total_s), pos="top")]
    if spec.text.strip():
        mid = max(0.0, total_s / 2 - 1.0)
        items.append(Title(text=spec.text, start=mid, dur=min(2.0, total_s - mid), pos="bottom"))
    if spec.ending.strip():
        start = max(0.0, total_s - 3.0)
        items.append(Title(text=spec.ending, start=start, dur=total_s - start, pos="center"))
    return Titles(items=tuple(items))


def write_titles(titles: Titles, out_dir: str) -> dict:
    """Write TITLES.json + TITLES.md together into out_dir. Returns a status dict."""
    Path(out_dir).mkdir(parents=True, exist_ok=True)
    tj, tm = Path(out_dir) / "TITLES.json", Path(out_dir) / "TITLES.md"
    titles.write(tj)
    tm.write_text(titles.to_markdown())
    return {"status": "written", "titles": str(tj), "md": str(tm), "count": len(titles.items)}


def main() -> None:
    ap = argparse.ArgumentParser(description="write TITLES.json + TITLES.md — the reel titles (§5.16)")
    sub = ap.add_subparsers(dest="cmd", required=True)
    b = sub.add_parser("build", help="draft TITLES from SPEC + reel length")
    b.add_argument("--spec", required=True, help="SPEC.json (from reel-brief) — hook/text/ending source")
    b.add_argument("--total", type=float, required=True, help="reel length, seconds (EDL sum / probed master)")
    b.add_argument("--out-dir", required=True, help="dir for TITLES.json + TITLES.md")
    a = ap.parse_args()
    titles = build_titles(Spec.from_json(a.spec), a.total)
    print(json.dumps(write_titles(titles, a.out_dir), indent=2))
    raise SystemExit(0)


if __name__ == "__main__":
    main()
```

def _frame_mean(path, t):
    # a crude luminance proxy: signalstats YAVG at time t
    out = proc.run(["ffmpeg", "-ss", str(t), "-i", str(path), "-vf", "signalstats,metadata=mode=print",
                    "-frames:v", "1", "-f", "null", "-"], check=False).stderr.decode("utf-8", "replace")
